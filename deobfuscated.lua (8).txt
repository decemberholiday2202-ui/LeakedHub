local function findCharacterBoundary(text, startIndex, characterSet)
	local textPrefix = ""

	for i = 1, #text do
		local quotient = (i - 1) / (#text - 1)
		local textPrefixNumber = math.floor((startIndex.R + (characterSet.R - startIndex.R) * quotient) * 255)
		local findCharacterBoundaryNumber = math.floor((startIndex.G + (characterSet.G - startIndex.G) * quotient) * 255)
		local number = math.floor((startIndex.B + (characterSet.B - startIndex.B) * quotient) * 255)

		textPrefix = textPrefix .. string.format("<font color=\"rgb(%d,%d,%d)\">%s</font>", textPrefixNumber, findCharacterBoundaryNumber, number, text:sub(i, i))
	end

	return textPrefix
end

local function updateInstanceProperties()
	local isUpdateInstancePropertiesSurvivorFlag = false

	isSendRemotesSurvivorData.killerModel = nil

	if workspace:FindFirstChild("Players") and workspace.Players:FindFirstChild("Killers") then
		for _, child in pairs(workspace.Players.Killers:GetChildren()) do
			if child:GetAttribute("Username") ~= game.Players.LocalPlayer.Name then
				if child:GetAttribute("Username") and game.Players:FindFirstChild(child:GetAttribute("Username")) then
					isSendRemotesSurvivorData.killerModel = child
				end
			else
				isSendRemotesSurvivorData.killerModel = child
				isUpdateInstancePropertiesSurvivorFlag = true
			end
		end

		isSendRemotesSurvivorData.isSurvivor = not isUpdateInstancePropertiesSurvivorFlag
	end
end
local function isSendRemoteEventValid(argument)
	local PlayerGui = game.Players.LocalPlayer.PlayerGui

	if PlayerGui:FindFirstChild("MainUI") and PlayerGui.MainUI:FindFirstChild("AbilityContainer") then
		local firstChild = PlayerGui.MainUI.AbilityContainer:FindFirstChild(argument)

		if firstChild and firstChild:FindFirstChild("CooldownTime") then
			return firstChild.CooldownTime.Text == ""
		end
	end

	return false
end

local function floor()
	local secondaryTimestamp = tick()
	local difference = secondaryTimestamp - timestamp

	if difference >= 1 then
		number = floorNumber / difference
		floorNumber = 0
		timestamp = secondaryTimestamp
	end

	return math.floor(number)
end

local function createBodyGyro()
	if not flyingData.flying then
		flyingData.flying = true

		local LocalPlayer = Players.LocalPlayer
		local humanoidRootPartContainer = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
		local HumanoidRootPart = humanoidRootPartContainer:WaitForChild("HumanoidRootPart")
		local Humanoid = humanoidRootPartContainer:FindFirstChildOfClass("Humanoid")

		if Humanoid then
			Humanoid.AutoRotate = false
			pData.bodyGyro = Instance.new("BodyGyro")
			pData.bodyGyro.P = 90000
			pData.bodyGyro.MaxTorque = Vector3.new(9000000000, 9000000000, 9000000000)
			pData.bodyGyro.CFrame = HumanoidRootPart.CFrame
			pData.bodyGyro.Parent = HumanoidRootPart
			pData.bodyVel = Instance.new("BodyVelocity")
			pData.bodyVel.MaxForce = Vector3.new(9000000000, 9000000000, 9000000000)
			pData.bodyVel.Velocity = Vector3.zero
			pData.bodyVel.Parent = HumanoidRootPart
			pData.flightConn = RunService.Heartbeat:Connect(function()
				local CurrentCamera = workspace.CurrentCamera
				local LookVector = CurrentCamera.CFrame.LookVector
				local RightVector = CurrentCamera.CFrame.RightVector
				local MoveDirection = Humanoid.MoveDirection
				local createBodyGyroNumber = MoveDirection:Dot(Vector3.new(LookVector.X, 0, LookVector.Z).Unit)
				local sumNumber = MoveDirection:Dot(Vector3.new(RightVector.X, 0, RightVector.Z).Unit)
				local unitNumber = 0

				if MoveDirection.Magnitude > 0 then
					unitNumber = LookVector.Y * createBodyGyroNumber
				end

				local sum = LookVector * createBodyGyroNumber + RightVector * sumNumber
				local unit = Vector3.new(sum.X, unitNumber, sum.Z)

				if unit.Magnitude > 1 then
					unit = unit.Unit
				end

				pData.bodyVel.Velocity = unit * flySpeedData.flySpeed
				pData.bodyGyro.CFrame = CFrame.lookAt(HumanoidRootPart.Position, HumanoidRootPart.Position + LookVector, CurrentCamera.CFrame.UpVector)
			end)

			return
		end

		return
	end
end
local function secondaryUpdateInstanceProperties()
	if flyingData.flying then
		flyingData.flying = false
		handleInstanceProperties()

		if pData.flightConn then
			pData.flightConn:Disconnect()
			pData.flightConn = nil
		end

		if pData.bodyGyro then
			pData.bodyGyro:Destroy()
			pData.bodyGyro = nil
		end

		if pData.bodyVel then
			pData.bodyVel:Destroy()
			pData.bodyVel = nil
		end

		local Character = Players.LocalPlayer.Character
		local updateInstancePropertiesCondition = Character and Character:FindFirstChildOfClass("Humanoid")

		if updateInstancePropertiesCondition then
			updateInstancePropertiesCondition.AutoRotate = true
		end

		return
	end
end
local function updateHeartbeatConnection(infiniteStamina)
	flyingData.infiniteStamina = infiniteStamina

	local Sprinting = require(game.ReplicatedStorage.Systems.Character.Game.Sprinting)

	if not flyingData.infiniteStamina then
		if pData.staminaConnection then
			pData.staminaConnection:Disconnect()
			pData.staminaConnection = nil
		end

		Sprinting.StaminaLossDisabled = nil
	else
		if pData.staminaConnection then
			pData.staminaConnection:Disconnect()
		end

		pData.staminaConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.infiniteStamina then
				function Sprinting.StaminaLossDisabled()
					return true
				end

				return
			end

			pData.staminaConnection:Disconnect()
			Sprinting.StaminaLossDisabled = nil
		end)
	end
end
local function alternateUpdateInstanceProperties()
	if not pData.antiStunConnection then
		pData.antiStunConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.antiStunActive then
				local Character = game.Players.LocalPlayer.Character

				if Character and Character:FindFirstChild("SpeedMultipliers") then
					local Stunned = Character.SpeedMultipliers:FindFirstChild("Stunned")

					if Stunned and Stunned.Value < 1 then
						Stunned.Value = 1
					end
				end

				return
			end
		end)

		return
	end
end
local function updateAntiStunConnection()
	if pData.antiStunConnection then
		pData.antiStunConnection:Disconnect()
		pData.antiStunConnection = nil
	end
end
local function additionalUpdateInstanceProperties()
	if not pData.antiSlowConnection then
		pData.antiSlowConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.antiSlowActive then
				local Character = game.Players.LocalPlayer.Character

				if Character and Character:FindFirstChild("SpeedMultipliers") then

						if child.Value < 1 then
							child.Value = 1
						end
					end
				end

				return
			end
		end)

		return
	end
end

local function createSntBillboard(parent, textLabel, textLabelColor, createSntBillboardFlag)
	if parent and parent:IsA("BasePart") then
		if not textLabelColor then
			textLabelColor = Color3.fromRGB(255, 255, 255)
		end

		local SNT_Billboard = parent:FindFirstChild("SNT_Billboard")

		if SNT_Billboard then
			SNT_Billboard:Destroy()
		end

		local BillboardGui = Instance.new("BillboardGui")

		BillboardGui.Name = "SNT_Billboard"
		BillboardGui.Adornee = parent
		BillboardGui.Size = UDim2.new(0, 200, 0, 50)
		BillboardGui.StudsOffset = createSntBillboardFlag or Vector3.new(0, 3, 0)
		BillboardGui.AlwaysOnTop = true
		BillboardGui.MaxDistance = 1000
		BillboardGui.Parent = parent

		local TextLabel = Instance.new("TextLabel")

		TextLabel.Size = UDim2.new(1, 0, 1, 0)
		TextLabel.BackgroundTransparency = 1
		TextLabel.Text = textLabel
		TextLabel.TextColor3 = textLabelColor
		TextLabel.TextSize = flySpeedData.espTextSize or 14
		TextLabel.Font = Enum.Font.GothamBold
		TextLabel.TextStrokeTransparency = 0
		TextLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
		TextLabel.Parent = BillboardGui

		return BillboardGui
	end

	return nil
end
local function createSntHighlight(parent, fillColor)
	if parent and parent:IsA("Model") then
		if not fillColor then
			fillColor = Color3.fromRGB(255, 255, 255)
		end

		local SNT_Highlight = parent:FindFirstChild("SNT_Highlight")

		if SNT_Highlight then
			SNT_Highlight:Destroy()
		end

		local Highlight = Instance.new("Highlight")

		Highlight.Name = "SNT_Highlight"
		Highlight.Adornee = parent
		Highlight.FillColor = fillColor
		Highlight.OutlineColor = fillColor
		Highlight.FillTransparency = flySpeedData.highlightTransparency or 0.5
		Highlight.OutlineTransparency = flySpeedData.highlightOutlineTransparency or 0
		Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		Highlight.Parent = parent

		return Highlight
	end

	return nil
end
local function secondaryFloor(floorNumber)
	local Character = game.Players.LocalPlayer.Character

	if Character then
		local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

		if HumanoidRootPart then
			return math.floor((floorNumber - HumanoidRootPart.Position).Magnitude)
		end

		return 0
	end

	return 0
end
local function handleText(textArgument)
	if updateTextData[textArgument] then
		for _, item in pairs(updateTextData[textArgument]) do
			if item and item.Parent then
				pcall(function()
					item:Destroy()
				end)
			end
		end

		updateTextData[textArgument] = {}
	end

	if textData[textArgument] then
		for _, item in pairs(textData[textArgument]) do
			if item then
				pcall(function()
					item:Disconnect()
				end)
			end
		end

		textData[textArgument] = {}
	end
end
local function updateText()
	handleText("survivors")

	if optionData.SurvivorsHighlight or (optionData.SurvivorsNameESP or optionData.SurvivorsDistance) then
		local players = workspace:FindFirstChild("Players")

		if players then
			local Survivors = players:FindFirstChild("Survivors")

			if Survivors then
				for _, child in pairs(Survivors:GetChildren()) do
					if child:IsA("Model") then
						local updateHeartbeatConnection = child:GetAttribute("Username") or child.Name

						if optionData.SurvivorsHighlight then
							local updateTextCondition = createSntHighlight(child, itemsData.Survivor)

							if updateTextCondition then
								table.insert(updateTextData.survivors, updateTextCondition)
							end
						end

						if optionData.SurvivorsNameESP or optionData.SurvivorsDistance then
							local Head = child:FindFirstChild("Head")

							if Head then
								local textLabel = optionData.SurvivorsNameESP and updateHeartbeatConnection or ""
								local updateTextCondition = createSntBillboard(Head, textLabel, itemsData.Survivor, Vector3.new(0, 4, 0))

								if updateTextCondition then
									table.insert(updateTextData.survivors, updateTextCondition)

									if optionData.SurvivorsDistance then
										local connection = game:GetService("RunService").Heartbeat:Connect(function()
											if updateTextCondition and updateTextCondition.Parent and Head and Head.Parent then
												local updateText = secondaryFloor(Head.Position)

												updateTextCondition.TextLabel.Text = updateHeartbeatConnection .. " [" .. updateText .. "m]"
											end
										end)

										table.insert(textData.survivors, connection)
									end
								end
							end
						end
					end
				end

				return
			end

			return
		end

		return
	end
end

local function alternateUpdateHeartbeatConnection()
	handleText("items")

	if optionData.ItemsHighlight or (optionData.ItemsNameESP or optionData.ItemsDistance) then
		local function updateHeartbeatConnection(item)
			for _, child in pairs(item:GetChildren()) do
				if not child:IsA("Tool") then
					if child:IsA("Folder") or child:IsA("Model") then
						updateHeartbeatConnection(child)
					end
				else
					local parent = child:FindFirstChild("Handle") or child:FindFirstChildWhichIsA("BasePart")

					if parent then
						if optionData.ItemsHighlight then
							local updateTextCondition = createSntHighlight(child, itemsData.Items)

							if updateTextCondition then
								table.insert(updateTextData.items, updateTextCondition)
							end
						end

						if optionData.ItemsNameESP or optionData.ItemsDistance then
							local textLabel = optionData.ItemsNameESP and child.Name or ""
							local updateTextCondition = createSntBillboard(parent, textLabel, itemsData.Items, Vector3.new(0, 2, 0))

							if updateTextCondition then
								table.insert(updateTextData.items, updateTextCondition)

								if optionData.ItemsDistance then
									local connection = game:GetService("RunService").Heartbeat:Connect(function()
										if updateTextCondition and updateTextCondition.Parent and parent and parent.Parent then
											local updateHeartbeatConnection = secondaryFloor(parent.Position)

											updateTextCondition.TextLabel.Text = child.Name .. " [" .. updateHeartbeatConnection .. "m]"
										end
									end)

									table.insert(textData.items, connection)
								end
							end
						end
					end
				end
			end
		end

		local updateHeartbeatConnectionData = {
			workspace,
			workspace:FindFirstChild("Map")
		}

		for _, item in pairs(updateHeartbeatConnectionData) do
			if item then
				updateHeartbeatConnection(item)
			end
		end

		return
	end
end
local function additionalUpdateHeartbeatConnection()
	handleText("generators")

	if optionData.GeneratorsHighlight or (optionData.GeneratorsNameESP or optionData.GeneratorsDistance) then
		local function updateHeartbeatConnection(item)
			for _, child in pairs(item:GetChildren()) do
				if not child.Name:lower():find("generator") and not child.Name:lower():find("gen") then
					if child:IsA("Folder") or child:IsA("Model") then
						updateHeartbeatConnection(child)
					end
				else
					local parent = child:FindFirstChildWhichIsA("BasePart") or child

					if optionData.GeneratorsHighlight then
						local updateTextCondition = createSntHighlight(child, itemsData.Generator)

						if updateTextCondition then
							table.insert(updateTextData.generators, updateTextCondition)
						end
					end

					if optionData.GeneratorsNameESP or optionData.GeneratorsDistance then
						local textLabel = optionData.GeneratorsNameESP and "Generator" or ""
						local updateTextCondition = createSntBillboard(parent, textLabel, itemsData.Generator, Vector3.new(0, 5, 0))

						if updateTextCondition then
							table.insert(updateTextData.generators, updateTextCondition)

							if optionData.GeneratorsDistance then
								local connection = game:GetService("RunService").Heartbeat:Connect(function()
									if updateTextCondition and updateTextCondition.Parent and parent and parent.Parent then
										local updateHeartbeatConnection = secondaryFloor(parent.Position)

										updateTextCondition.TextLabel.Text = "Generator" .. " [" .. updateHeartbeatConnection .. "m]"
									end
								end)

								table.insert(textData.generators, connection)
							end
						end
					end
				end
			end
		end

		local updateHeartbeatConnectionData = {
			workspace,
			workspace:FindFirstChild("Map")
		}

		for _, item in pairs(updateHeartbeatConnectionData) do
			if item then
				updateHeartbeatConnection(item)
			end
		end

		return
	end
end
local function handleHeartbeatConnection()
	pcall(function()
		updateText()
		secondaryUpdateHeartbeatConnection()
		alternateUpdateHeartbeatConnection()
		additionalUpdateHeartbeatConnection()
	end)
end

local function updateLastEsp()
	if connection then
		pcall(function()
			connection:Disconnect()
		end)
	end

	connection = game:GetService("RunService").Heartbeat:Connect(function()
		local lastEspUpdate = tick()

		if (flySpeedData.espUpdateInterval or 1) <= lastEspUpdate - (lastAuraUseData.lastEspUpdate or 0) then
			lastAuraUseData.lastEspUpdate = lastEspUpdate
			handleHeartbeatConnection()
		end
	end)
end

local function handleCondition(conditionFlag, flag)
	local childs = {}

	if not conditionFlag then
		if not flag then
			local Survivors = workspace.Players:FindFirstChild("Survivors")

			if Survivors then
				for _, child in pairs(Survivors:GetChildren()) do
					if child:IsA("Model") and child:FindFirstChild("Humanoid") and child.Humanoid.Health > 0 then
						table.insert(childs, child)
					end
				end
			end
		else
			local Killers = workspace.Players:FindFirstChild("Killers")

			if Killers then
				for _, child in pairs(Killers:GetChildren()) do
					if child:IsA("Model") and child:FindFirstChild("Humanoid") and child.Humanoid.Health > 0 then
						table.insert(childs, child)
					end
				end
			end
		end
	else
		local Survivors = workspace.Players:FindFirstChild("Survivors")

		if Survivors then
			for _, child in pairs(Survivors:GetChildren()) do
				if child:IsA("Model") and child:FindFirstChild("Humanoid") and child.Humanoid.Health > 0 then
					table.insert(childs, child)
				end
			end
		end
	end

	if #childs ~= 0 then
		local Character = game.Players.LocalPlayer.Character

		if Character then
			local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				local humanoidRootPartContainer = nil
				local huge = math.huge

				for _, item in pairs(childs) do
					local humanoidRootPart = item:FindFirstChild("HumanoidRootPart")

					if humanoidRootPart then
						local Magnitude = (humanoidRootPart.Position - HumanoidRootPart.Position).Magnitude

						if Magnitude < huge then
							huge = Magnitude
							humanoidRootPartContainer = item
						end
					end
				end

				return humanoidRootPartContainer
			end

			return nil
		end

		return nil
	end

	return nil
end
local function updateCFrame(humanoidRootPartContainer, updateCFrameFlag)
	if humanoidRootPartContainer and humanoidRootPartContainer:FindFirstChild("HumanoidRootPart") then
		local Character = game.Players.LocalPlayer.Character

		if Character then
			local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				local humanoidRootPart = humanoidRootPartContainer.HumanoidRootPart
				local updateCFrameNumber = tick()
				local updateCFrameOption = updateCFrameFlag or 2

				while updateCFrameOption > tick() - updateCFrameNumber and humanoidRootPartContainer and humanoidRootPartContainer.Parent and humanoidRootPartContainer:FindFirstChild("HumanoidRootPart") do
					workspace.CurrentCamera.CFrame = CFrame.new(workspace.CurrentCamera.CFrame.Position, humanoidRootPart.Position)
					HumanoidRootPart.CFrame = CFrame.lookAt(HumanoidRootPart.Position, Vector3.new(humanoidRootPart.Position.X, humanoidRootPart.Position.Y, humanoidRootPart.Position.Z))
					task.wait()
				end

				return
			end

			return
		end

		return
	end
end
local function updateOnClientEventConnection(searchableText, secondarySearchableText, item)
	if secondaryData[searchableText] then
		secondaryData[searchableText]:Disconnect()
	end

	secondaryData[searchableText] = game.ReplicatedStorage.Modules.Network.RemoteEvent.OnClientEvent:Connect(function(...)
		local searchQueryData = { ... }

		if #searchQueryData >= 2 and searchQueryData[1] == "UseActorAbility" then
			local searchQuery = searchQueryData[2]

			if table.find(secondarySearchableText, searchQuery) then
				local humanoidRootPartContainer = handleCondition(string.find(searchableText, "Killer") ~= nil, item)

				if humanoidRootPartContainer then
					updateCFrame(humanoidRootPartContainer, 1.5)
				end
			end
		end
	end)
end
local function secondaryHandler(searchableText, flag)
	noliData[searchableText] = flag

	if not flag then
		if secondaryData[searchableText] then
			secondaryData[searchableText]:Disconnect()
			secondaryData[searchableText] = nil
		end
	else
		local condition = conditionData[searchableText]

		if condition then
			if type(condition.targetKiller) ~= "table" then
				updateOnClientEventConnection(searchableText, condition.abilities, condition.targetKiller)
			else
				for k, item in pairs(condition.targetKiller) do
					updateOnClientEventConnection(searchableText, { k }, item)
				end
			end
		end
	end
end

game.Players.LocalPlayer.CharacterAdded:Connect(function(character)
	task.wait(1)

	local _ = character.Name

	for k, item in pairs(noliData) do
		if item then
			secondaryHandler(k, false)
		end
	end
end)

local function findFirstChild(inputCondition, secondaryArgument)
	if inputCondition and inputCondition:IsA("Instance") then
		return inputCondition:FindFirstChild(secondaryArgument)
	end

	return nil
end
local function secondaryHandleInstanceProperties()
	local LocalPlayer = game.Players.LocalPlayer
	local instancePropertiesCondition = findFirstChild(workspace.Players, "Survivors")

	if instancePropertiesCondition then
		for _, child in pairs(instancePropertiesCondition:GetChildren()) do
			if child:IsA("Model") then
				local Username = child:GetAttribute("Username")

				if Username and Username == LocalPlayer.Name then
					return true
				end
			end
		end

		return false
	end

	return false
end
local function handleData()
	local childs = {}
	local dataCondition = findFirstChild(workspace.Map, "Ingame")

	if dataCondition then
		local condition = findFirstChild(dataCondition, "Map")

		if condition then
			for _, child in pairs(condition:GetChildren()) do
				if child.Name == "Generator" then
					table.insert(childs, child)
				end
			end
		end
	end

	return childs
end
local function getData(item)
	local getDataFlag = findFirstChild(item, "Progress")

	if not getDataFlag or not getDataFlag:IsA("NumberValue") then
		return 0
	end

	return getDataFlag.Value
end
local function secondaryHandleData()
	local dataResult = handleData()
	local vs = {}

	for _, item in pairs(dataResult) do
		if getData(item) < 100 then
			table.insert(vs, item)
		end
	end

	return vs
end
local function getInstancePropertiesPosition(item)
	local inputCondition = findFirstChild(item, "Instances")

	if inputCondition then
		local input = findFirstChild(inputCondition, "Cube.001")

		if input and input:IsA("BasePart") then
			return input.Position
		end
	end

	local condition = findFirstChild(item, "Positions")

	if condition then
		local input = findFirstChild(condition, "Center")

		if input and input:IsA("BasePart") then
			return input.Position
		end
	end

	for _, descendant in pairs(item:GetDescendants()) do
		if descendant:IsA("BasePart") then
			return descendant.Position
		end
	end

	return Vector3.new(0, 0, 0)
end
local function handleHumanoidRootPartContainer()
	local LocalPlayer = game.Players.LocalPlayer
	local Survivors = workspace.Players:FindFirstChild("Survivors")

	if Survivors then
		for _, child in pairs(Survivors:GetChildren()) do
			if child:IsA("Model") then
				local Username = child:GetAttribute("Username")

				if Username and Username == LocalPlayer.Name then
					return child
				end
			end
		end

		return nil
	end

	return nil
end
local function handleCurrentRepairGenerator()
	local inputCondition = handleHumanoidRootPartContainer()

	if inputCondition then
		local input = findFirstChild(inputCondition, "HumanoidRootPart")

		if input then
			local secondaryData = secondaryHandleData()

			if #secondaryData ~= 0 then
				local item = nil
				local huge = math.huge

				for _, secondaryItem in pairs(secondaryData) do
					local magnitudeNumber = getInstancePropertiesPosition(secondaryItem)

					if magnitudeNumber ~= Vector3.new(0, 0, 0) then
						local Magnitude = (magnitudeNumber - input.Position).Magnitude

						if Magnitude < huge then
							huge = Magnitude
							item = secondaryItem
						end
					end
				end

				return item
			end

			return nil
		end

		return nil
	end

	return nil
end
local function secondaryUpdateCFrame(sumNumber)
	local updateCFrameCondition = handleHumanoidRootPartContainer()

	if updateCFrameCondition then
		local cFrameCondition = findFirstChild(updateCFrameCondition, "HumanoidRootPart")

		if cFrameCondition then
			local updateCFrameNumber = sumNumber + Vector3.new(0, 0, 0)

			cFrameCondition.CFrame = CFrame.new(updateCFrameNumber)

			return true
		end

		return false
	end

	return false
end
local function updatePromptActivated(inputCondition)
	if not flyingData.promptActivated then
		local updatePromptActivatedCondition = findFirstChild(inputCondition, "Main")

		if updatePromptActivatedCondition then
			local secondaryDescendant = findFirstChild(updatePromptActivatedCondition, "Prompt")

			if not secondaryDescendant then
				for _, descendant in pairs(updatePromptActivatedCondition:GetDescendants()) do
					if descendant:IsA("ProximityPrompt") then
						secondaryDescendant = descendant

						break
					end
				end
			end

			if secondaryDescendant and secondaryDescendant:IsA("ProximityPrompt") then
				fireproximityprompt(secondaryDescendant)
				flyingData.promptActivated = true

				return true
			end

			return false
		end

		return false
	end

	return true
end
local function sendRemotes(inputCondition)
	local remoteEventCondition = findFirstChild(inputCondition, "Remotes")

	if remoteEventCondition then
		local remoteEvent = findFirstChild(remoteEventCondition, "RE")

		if not remoteEvent or not remoteEvent:IsA("RemoteEvent") or isSendRemotesSurvivorData.repairDebounce[inputCondition] then
			return false
		end

		isSendRemotesSurvivorData.repairDebounce[inputCondition] = true

		local ok = pcall(function()
			remoteEvent:FireServer()
		end)

		task.delay(3, function()
			isSendRemotesSurvivorData.repairDebounce[inputCondition] = nil
		end)

		return ok
	end

	return false
end
local function fallbackUpdateInstanceProperties()
	if flyingData.autoRepairActive then
		flyingData.autoRepairActive = false

		if pData.autoRepairConnection then
			pData.autoRepairConnection:Disconnect()
			pData.autoRepairConnection = nil
		end

		isSendRemotesSurvivorData.currentRepairGenerator = nil
		isSendRemotesSurvivorData.repairDebounce = {}
		flyingData.promptActivated = false
		flyingData.teleportedToGenerator = false

		return
	end
end
local function nestedUpdateInstanceProperties()
	if not flyingData.autoRepairActive then
		flyingData.autoRepairActive = true
		pData.autoRepairConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.autoRepairActive then
				if secondaryHandleInstanceProperties() then
					if handleHumanoidRootPartContainer() then
						if isSendRemotesSurvivorData.currentRepairGenerator and (not isSendRemotesSurvivorData.currentRepairGenerator.Parent or getData(isSendRemotesSurvivorData.currentRepairGenerator) >= 100) then
							isSendRemotesSurvivorData.currentRepairGenerator = nil
							flyingData.promptActivated = false
							flyingData.teleportedToGenerator = false
							lastAuraUseData.teleportTime = 0
						end

						if not isSendRemotesSurvivorData.currentRepairGenerator then
							isSendRemotesSurvivorData.currentRepairGenerator = handleCurrentRepairGenerator()

							if not isSendRemotesSurvivorData.currentRepairGenerator then
								fallbackUpdateInstanceProperties()

								return
							end

							flyingData.promptActivated = false
							flyingData.teleportedToGenerator = false
							lastAuraUseData.teleportTime = 0
						end

						if not flyingData.teleportedToGenerator then
							local updateCFrameNumber = getInstancePropertiesPosition(isSendRemotesSurvivorData.currentRepairGenerator)

							if updateCFrameNumber ~= Vector3.new(0, 0, 0) then
								secondaryUpdateCFrame(updateCFrameNumber)
								flyingData.teleportedToGenerator = true
								lastAuraUseData.teleportTime = tick()
							end
						end

						if not flyingData.promptActivated and flyingData.teleportedToGenerator and tick() - lastAuraUseData.teleportTime >= 0.5 then
							updatePromptActivated(isSendRemotesSurvivorData.currentRepairGenerator)
							flyingData.promptActivated = true
						end

						if flyingData.promptActivated then
							sendRemotes(isSendRemotesSurvivorData.currentRepairGenerator)
						end

						if getData(isSendRemotesSurvivorData.currentRepairGenerator) >= 100 then
							task.wait(flySpeedData.teleportCooldown)
							isSendRemotesSurvivorData.currentRepairGenerator = nil
							flyingData.promptActivated = false
							flyingData.teleportedToGenerator = false
							lastAuraUseData.teleportTime = 0
						end

						task.wait(flySpeedData.repairCheckInterval)

						return
					end

					fallbackUpdateInstanceProperties()

					return
				end

				fallbackUpdateInstanceProperties()

				return
			end
		end)

		return
	end
end
local function handleSendRemoteEvent()
	if not (tick() - lastAuraUseData.lastFriedChickenTime < flySpeedData.friedChickenCooldown) then
		return true
	end

	return false
end
local function sendRemoteEvent()
	if handleSendRemoteEvent() then
		if not pcall(function()
			game.ReplicatedStorage.Modules.Network.RemoteEvent:FireServer(unpack({
				"UseActorAbility",
				"FriedChicken"
			}))
		end) then
			return false
		end

		lastAuraUseData.lastFriedChickenTime = tick()

		return true
	end

	return false
end

	end

local function alternateHandleData(item)
	local descendants = {}
	local secondaryData = {
		workspace.Map.Ingame,
		workspace.Map.Ingame.Map,
		workspace.Map
	}

	for _, secondaryItem in pairs(secondaryData) do
		if secondaryItem then
			local firstChild = secondaryItem:FindFirstChild(item)

			if firstChild and firstChild:FindFirstChild("ItemRoot") then
				table.insert(descendants, firstChild)
			end

			for _, descendant in pairs(secondaryItem:GetDescendants()) do
				if item == descendant.Name and descendant:FindFirstChild("ItemRoot") then
					table.insert(descendants, descendant)
				end
			end
		end
	end

	return descendants
end
local function getCFramePosition(itemRootContainer)
	local ItemRoot = itemRootContainer:FindFirstChild("ItemRoot")

	if not ItemRoot or not ItemRoot:IsA("BasePart") then
		for _, descendant in pairs(itemRootContainer:GetDescendants()) do
			if descendant:IsA("BasePart") then
				return descendant.Position
			end
		end

		return nil
	end

	return ItemRoot.Position
end
local function secondaryHandleCondition(itemRootContainer)
	if itemRootContainer then
		local ItemRoot = itemRootContainer:FindFirstChild("ItemRoot")

		if ItemRoot then
			local conditionOption = ItemRoot:FindFirstChildOfClass("ProximityPrompt") or itemRootContainer:FindFirstChildOfClass("ProximityPrompt")

			if not conditionOption or not conditionOption:IsA("ProximityPrompt") then
				return false
			end

			fireproximityprompt(conditionOption)

			return true
		end

		return false
	end

	return false
end

				HumanoidRootPart.CFrame = CFrame.new(sum)

				return true
			end

			return false
		end

		return false
	end

	return false
end

local function handleCurrentTarget()
	local Character = game.Players.LocalPlayer.Character

	if Character then
		local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

		if HumanoidRootPart then
			local Survivors = workspace.Players:FindFirstChild("Survivors")

			if Survivors then
				local secondaryChild = nil
				local huge = math.huge

				for _, child in pairs(Survivors:GetChildren()) do
					if child:IsA("Model") and child:FindFirstChild("Humanoid") and child.Humanoid.Health > 0 then
						local humanoidRootPart = child:FindFirstChild("HumanoidRootPart")

						if humanoidRootPart then
							local Magnitude = (humanoidRootPart.Position - HumanoidRootPart.Position).Magnitude

							if Magnitude < huge then
								huge = Magnitude
								secondaryChild = child
							end
						end
					end
				end

				return secondaryChild
			end

			return nil
		end

		return nil
	end

	return nil
end
local function additionalUpdateCFrame(humanoidRootPartContainer)
	if humanoidRootPartContainer and humanoidRootPartContainer:FindFirstChild("HumanoidRootPart") then
		local Character = game.Players.LocalPlayer.Character

		if Character then
			local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

			if HumanoidRootPart then
				local humanoidRootPart = humanoidRootPartContainer.HumanoidRootPart

				if not flyingData.killAllTeleport then
					if not flyingData.killAllFly then
						Character.Humanoid:MoveTo(humanoidRootPart.Position)
					else
						if not flyingData.flying then
							createBodyGyro()
						end

						local Unit = (humanoidRootPart.Position - HumanoidRootPart.Position).Unit

						Character.Humanoid:MoveTo(HumanoidRootPart.Position + Unit * 10)
					end
				else
					local LookVector = humanoidRootPart.CFrame.LookVector
					local updateCFrameNumber = humanoidRootPart.Position - LookVector * 2.7 + Vector3.new(0, 1.5, 0)

					HumanoidRootPart.CFrame = CFrame.new(updateCFrameNumber)
					HumanoidRootPart.CFrame = CFrame.lookAt(updateCFrameNumber, humanoidRootPart.Position)
				end

				return
			end

			return
		end

		return
	end
end
local function handleResult()
	local Character = game.Players.LocalPlayer.Character

	if Character then
		local CharacterName = Character.Name

		if CharacterName ~= "Noli" then
			if CharacterName ~= "c001kidd" then
				return "Slash"
			end

			return "Punch"
		end

		return "Stab"
	end

	return "Slash"
end
local function secondarySendRemoteEvent(sendRemoteEventArgument)
	if not isSendRemotesSurvivorData.skillCooldowns[sendRemoteEventArgument] or not (tick() - isSendRemotesSurvivorData.skillCooldowns[sendRemoteEventArgument] < 5) then
		game.ReplicatedStorage.Modules.Network.RemoteEvent:FireServer(unpack({
			"UseActorAbility",
			sendRemoteEventArgument
		}))
		isSendRemotesSurvivorData.skillCooldowns[sendRemoteEventArgument] = tick()

		return true
	end

	return false
end
local function alternateSendRemoteEvent()
	local sendRemoteEventResult = handleResult()

	game.ReplicatedStorage.Modules.Network.RemoteEvent:FireServer(unpack({
		"UseActorAbility",
		sendRemoteEventResult
	}))
end

local function updateInstancePropertiesVariantH()
	if not flyingData.killAllActive then
		flyingData.killAllActive = true

		local LocalPlayer = game.Players.LocalPlayer
		local _ = LocalPlayer.Character

		pData.killAllConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.killAllActive then
				if isSendRemotesSurvivorData.currentTarget and (not isSendRemotesSurvivorData.currentTarget.Parent or not isSendRemotesSurvivorData.currentTarget:FindFirstChild("Humanoid") or isSendRemotesSurvivorData.currentTarget.Humanoid.Health <= 0) then
					isSendRemotesSurvivorData.currentTarget = nil
				end

				if not isSendRemotesSurvivorData.currentTarget then
					isSendRemotesSurvivorData.currentTarget = handleCurrentTarget()

					if not isSendRemotesSurvivorData.currentTarget then
						return
					end
				end

				additionalUpdateCFrame(isSendRemotesSurvivorData.currentTarget)

				local Character = LocalPlayer.Character

				if Character and Character:FindFirstChild("HumanoidRootPart") and isSendRemotesSurvivorData.currentTarget:FindFirstChild("HumanoidRootPart") then
					local Magnitude = (isSendRemotesSurvivorData.currentTarget.HumanoidRootPart.Position - Character.HumanoidRootPart.Position).Magnitude

					if not flyingData.killAllTeleport then
						if Magnitude < 15 then
							alternateSendRemoteEvent()
						end
					else
						alternateSendRemoteEvent()
						task.wait(0.3)
					end
				end

				return
			end
		end)

		return
	end
end

	end
end

local function handleFlag()
	local secondaryChild = nil
	local huge = math.huge
	local Character = game.Players.LocalPlayer.Character

	if Character and Character:FindFirstChild("HumanoidRootPart") then
		local Survivors = workspace.Players:FindFirstChild("Survivors")

		if Survivors then
			for _, child in pairs(Survivors:GetChildren()) do
				if child:IsA("Model") and child:FindFirstChild("HumanoidRootPart") then
					local Magnitude = (Character.HumanoidRootPart.Position - child.HumanoidRootPart.Position).Magnitude

					if Magnitude < huge then
						huge = Magnitude
						secondaryChild = child
					end
				end
			end

			return secondaryChild, huge
		end

		return nil
	end

	return nil
end
local function updateLastAuraUse(sendRemoteEventText)
	local lastAuraUse = tick()

	if not (lastAuraUse - lastAuraUseData.lastAuraUse >= flySpeedData.auraCooldown) then
		return false
	end

	game.ReplicatedStorage.Modules.Network.RemoteEvent:FireServer("UseActorAbility", sendRemoteEventText)
	lastAuraUseData.lastAuraUse = lastAuraUse

	return true
end
local function updateAuraConnection()
	if not pData.auraConnection then
		pData.auraConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.killerAuraActive and isSendRemotesSurvivorData.isKiller then
				local updateAuraConnectionFlag, updateAuraConnectionResult = handleFlag()

				if updateAuraConnectionFlag and not (updateAuraConnectionResult > flySpeedData.auraRange) then
					local CharacterName = game.Players.LocalPlayer.Character.Name
					local updateAuraConnectionText = "Slash"

					if CharacterName ~= "Noli" then
						if CharacterName ~= "c001kidd" then
							if CharacterName == "Guest1337" then
								updateAuraConnectionText = "Punch"
							end
						else
							updateAuraConnectionText = "Punch"
						end
					else
						updateAuraConnectionText = "Stab"
					end

					updateLastAuraUse(updateAuraConnectionText)

					return
				end

				return
			end
		end)

		return
	end
end
local function secondaryUpdateAuraConnection()
	if not pData.auraConnection then
		pData.auraConnection = game:GetService("RunService").Heartbeat:Connect(function()
			if flyingData.slashAuraActive and isSendRemotesSurvivorData.isKiller then
				local updateAuraConnectionFlag, updateAuraConnectionResult = handleFlag()

				if updateAuraConnectionFlag and not (updateAuraConnectionResult > flySpeedData.auraRange) then
					updateLastAuraUse("Slash")

					return
				end

				return
			end
		end)

		return
	end
end
local function alternateUpdateAuraConnection()
	if pData.auraConnection and type(pData.auraConnection) == "userdata" then
		pcall(function()
			pData.auraConnection:Disconnect()
		end)
		pData.auraConnection = nil
	end
end
local function additionalHandler(argument, condition)
	alternateUpdateAuraConnection()

	if condition then
		if not isSendRemotesSurvivorData.isKiller then
			return false
		end

		if argument ~= "killer" then
			if argument == "slash" then
				secondaryUpdateAuraConnection()
			end
		else
			updateAuraConnection()
		end
	end
end
